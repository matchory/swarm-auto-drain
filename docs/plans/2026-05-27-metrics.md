## How to read this plan

> **For agentic workers:** Use the `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` sub-skill to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an OpenTelemetry-based metrics surface to the coordinator: domain metrics for lock/drain/role/forward/identity, RED HTTP metrics, and Go runtime metrics. Exposed via a Prometheus `/metrics` endpoint on a dedicated listener and/or via OTLP HTTP push. Both readers default off; existing deployments are unaffected.

**Architecture:** A new `internal/metrics` package owns a single `sdkmetric.MeterProvider` with up to two readers (Prometheus exporter, OTLP push exporter). It exposes a `Metrics` struct of typed instrument handles, grouped by subsystem, that domain packages take as a constructor parameter. A `noop` provider is used when neither reader is configured, so recording calls are always safe.

**Tech Stack:** Go 1.25, OpenTelemetry Go SDK v1.43.x (already a transitive dep), new direct deps: `otel/exporters/prometheus`, `otel/exporters/otlp/otlpmetric/otlpmetrichttp`, `contrib/instrumentation/runtime`.

Each task ends in a green `go test ./...` and a commit. Tasks 8 (HTTP middleware) and 10–14 (per-subsystem instrumentation) are independent of each other and can be executed in any order after Task 7.

## Testing Approach

- **Unit tests** drive each instrumented branch and assert the right instrument was incremented with the right attributes, using `sdkmetric.NewManualReader` and a small assertion helper in `internal/metrics/testing.go`.
- **HTTP middleware tests** verify route templating, status capture, and pass-through transparency.
- **Integration smoke** in `tests/integration/tests.sh` curls `/metrics` and greps for `swarm_auto_drain_lock_acquire_total`.
- The OTLP push path is exercised by the OTel SDK's own tests; we do not test it in CI.

// Package metrics owns the coordinator's OpenTelemetry metric provider
// and exports the typed instrument handles that domain packages record
// against. See docs/specs/2026-05-27-metrics-design.md.
package metrics

import (
	"fmt"

	"go.opentelemetry.io/otel/metric"
)

// Metrics is the full set of instruments exposed by the coordinator.
// Each sub-struct corresponds to a subsystem; domain packages receive
// only the sub-struct they need so the dependency surface stays narrow.
type Metrics struct {
	Lock     LockMetrics
	Drain    DrainMetrics
	Role     RoleMetrics
	Forward  ForwardMetrics
	Identity IdentityMetrics
	HTTP     HTTPMetrics
}

// LockMetrics covers cluster-mutex outcomes.
type LockMetrics struct {
	Acquire metric.Int64Counter      // attrs: outcome
	Release metric.Int64Counter      // attrs: outcome
	Steals  metric.Int64Counter      // attrs: reason
	Hold    metric.Float64Histogram  // unit: seconds
	Held    metric.Int64Gauge
}

// DrainMetrics covers node-drain outcomes and duration.
type DrainMetrics struct {
	Total          metric.Int64Counter      // attrs: outcome
	Duration       metric.Float64Histogram  // unit: seconds
	TasksRemaining metric.Int64Gauge
	QuorumCheck    metric.Int64Counter // attrs: result
}

// RoleMetrics covers the role detector's view of swarm role.
type RoleMetrics struct {
	Current     metric.Int64Gauge   // 0=worker, 1=manager
	Refresh     metric.Int64Counter // attrs: trigger, outcome
	Transitions metric.Int64Counter // attrs: from, to
}

// ForwardMetrics covers worker→manager proxy behaviour.
type ForwardMetrics struct {
	Attempts           metric.Int64Counter      // attrs: outcome
	Duration           metric.Float64Histogram  // unit: seconds
	CacheInvalidations metric.Int64Counter
}

// IdentityMetrics covers caller-identity resolution.
type IdentityMetrics struct {
	Resolutions metric.Int64Counter // attrs: source, outcome
}

// HTTPMetrics covers per-request RED metrics.
type HTTPMetrics struct {
	Requests metric.Int64Counter      // attrs: method, route, status
	Duration metric.Float64Histogram  // attrs: method, route; unit: seconds
}

// build wires a Metrics struct from a meter provider. Used by both the
// production constructor (New) and the test/noop constructors.
func build(meter metric.Meter) (Metrics, error) {
	var m Metrics
	var err error
	if m.Lock.Acquire, err = meter.Int64Counter("swarm_auto_drain.lock.acquire", metric.WithDescription("Lock acquire attempts by outcome.")); err != nil {
		return Metrics{}, fmt.Errorf("lock.acquire: %w", err)
	}
	if m.Lock.Release, err = meter.Int64Counter("swarm_auto_drain.lock.release", metric.WithDescription("Lock release attempts by outcome.")); err != nil {
		return Metrics{}, fmt.Errorf("lock.release: %w", err)
	}
	if m.Lock.Steals, err = meter.Int64Counter("swarm_auto_drain.lock.steals", metric.WithDescription("Stale-lock steals by reason.")); err != nil {
		return Metrics{}, fmt.Errorf("lock.steals: %w", err)
	}
	if m.Lock.Hold, err = meter.Float64Histogram("swarm_auto_drain.lock.hold", metric.WithUnit("s"), metric.WithDescription("Lock hold duration in seconds.")); err != nil {
		return Metrics{}, fmt.Errorf("lock.hold: %w", err)
	}
	if m.Lock.Held, err = meter.Int64Gauge("swarm_auto_drain.lock.held", metric.WithDescription("1 if this replica currently holds a lock, 0 otherwise.")); err != nil {
		return Metrics{}, fmt.Errorf("lock.held: %w", err)
	}

	if m.Drain.Total, err = meter.Int64Counter("swarm_auto_drain.drain.total", metric.WithDescription("Drain attempts by outcome.")); err != nil {
		return Metrics{}, fmt.Errorf("drain.total: %w", err)
	}
	if m.Drain.Duration, err = meter.Float64Histogram("swarm_auto_drain.drain.duration", metric.WithUnit("s"), metric.WithDescription("Drain duration in seconds.")); err != nil {
		return Metrics{}, fmt.Errorf("drain.duration: %w", err)
	}
	if m.Drain.TasksRemaining, err = meter.Int64Gauge("swarm_auto_drain.drain.tasks_remaining", metric.WithDescription("Tasks remaining on the node during an active drain.")); err != nil {
		return Metrics{}, fmt.Errorf("drain.tasks_remaining: %w", err)
	}
	if m.Drain.QuorumCheck, err = meter.Int64Counter("swarm_auto_drain.drain.quorum_check", metric.WithDescription("Pre-drain quorum checks by result.")); err != nil {
		return Metrics{}, fmt.Errorf("drain.quorum_check: %w", err)
	}

	if m.Role.Current, err = meter.Int64Gauge("swarm_auto_drain.role", metric.WithDescription("Current role: 0=worker, 1=manager.")); err != nil {
		return Metrics{}, fmt.Errorf("role: %w", err)
	}
	if m.Role.Refresh, err = meter.Int64Counter("swarm_auto_drain.role.refresh", metric.WithDescription("Role detector refresh outcomes.")); err != nil {
		return Metrics{}, fmt.Errorf("role.refresh: %w", err)
	}
	if m.Role.Transitions, err = meter.Int64Counter("swarm_auto_drain.role.transitions", metric.WithDescription("Role transitions.")); err != nil {
		return Metrics{}, fmt.Errorf("role.transitions: %w", err)
	}

	if m.Forward.Attempts, err = meter.Int64Counter("swarm_auto_drain.forward.attempts", metric.WithDescription("Forward attempts by outcome.")); err != nil {
		return Metrics{}, fmt.Errorf("forward.attempts: %w", err)
	}
	if m.Forward.Duration, err = meter.Float64Histogram("swarm_auto_drain.forward.duration", metric.WithUnit("s"), metric.WithDescription("Forward request duration including retries.")); err != nil {
		return Metrics{}, fmt.Errorf("forward.duration: %w", err)
	}
	if m.Forward.CacheInvalidations, err = meter.Int64Counter("swarm_auto_drain.forward.cache_invalidations", metric.WithDescription("Manager-URL cache invalidations.")); err != nil {
		return Metrics{}, fmt.Errorf("forward.cache_invalidations: %w", err)
	}

	if m.Identity.Resolutions, err = meter.Int64Counter("swarm_auto_drain.identity.resolutions", metric.WithDescription("Identity resolutions by source and outcome.")); err != nil {
		return Metrics{}, fmt.Errorf("identity.resolutions: %w", err)
	}

	if m.HTTP.Requests, err = meter.Int64Counter("swarm_auto_drain.http.requests", metric.WithDescription("HTTP requests by method/route/status.")); err != nil {
		return Metrics{}, fmt.Errorf("http.requests: %w", err)
	}
	if m.HTTP.Duration, err = meter.Float64Histogram("swarm_auto_drain.http.duration", metric.WithUnit("s"), metric.WithDescription("HTTP request duration.")); err != nil {
		return Metrics{}, fmt.Errorf("http.duration: %w", err)
	}

	return m, nil
}
```

// NewNoop returns a Metrics whose instrument handles are backed by an
// OpenTelemetry noop provider. Recording is cheap and goes nowhere.
// Used when no readers are configured and as the default in tests that
// don't care about metric values.
func NewNoop() Metrics {
	m, err := build(noop.NewMeterProvider().Meter("swarm-auto-drain"))
	if err != nil {
		// Noop provider does not return errors from instrument
		// creation; if build ever does, that's a programmer bug we
		// want to find immediately.
		panic("metrics: build against noop provider failed: " + err.Error())
	}
	return m
}

		}

package metrics

import (
	"context"
	"fmt"
	"net/http"

	otelprom "go.opentelemetry.io/otel/exporters/prometheus"
	"go.opentelemetry.io/otel/metric/noop"
	sdkmetric "go.opentelemetry.io/otel/sdk/metric"
	"go.opentelemetry.io/otel/sdk/resource"
	semconv "go.opentelemetry.io/otel/semconv/v1.27.0"
)

const meterName = "swarm-auto-drain"

// ShutdownFunc flushes both exporters with a caller-supplied budget.
type ShutdownFunc func(context.Context) error

// New constructs the metric provider, attaches whichever readers Config
// asks for, and returns the resulting Metrics, the HTTP handler for the
// Prometheus exporter (nil if disabled), and a shutdown func.
func New(ctx context.Context, cfg Config) (Metrics, http.Handler, ShutdownFunc, error) {
	if cfg.ListenAddr == "" && cfg.OTLPEndpoint == "" {
		m, err := build(noop.NewMeterProvider().Meter(meterName))
		if err != nil {
			return Metrics{}, nil, nil, fmt.Errorf("build noop: %w", err)
		}
		return m, nil, nil, nil
	}

	res, err := buildResource(cfg)
	if err != nil {
		return Metrics{}, nil, nil, fmt.Errorf("resource: %w", err)
	}

	providerOpts := []sdkmetric.Option{sdkmetric.WithResource(res)}

	var promHandler http.Handler
	if cfg.ListenAddr != "" {
		promReader, promErr := otelprom.New()
		if promErr != nil {
			return Metrics{}, nil, nil, fmt.Errorf("prometheus exporter: %w", promErr)
		}
		providerOpts = append(providerOpts, sdkmetric.WithReader(promReader))
		promHandler = promhttp.Handler() // wired in Step 4 below — placeholder kept for clarity
	}

	provider := sdkmetric.NewMeterProvider(providerOpts...)
	m, err := build(provider.Meter(meterName))
	if err != nil {
		return Metrics{}, nil, nil, fmt.Errorf("build instruments: %w", err)
	}
	shutdown := func(sCtx context.Context) error {
		if sErr := provider.Shutdown(sCtx); sErr != nil {
			return fmt.Errorf("metric provider shutdown: %w", sErr)
		}
		return nil
	}
	return m, promHandler, shutdown, nil
}

func buildResource(cfg Config) (*resource.Resource, error) {
	attrs := []resource.Option{
		resource.WithAttributes(
			semconv.ServiceName(meterName),
			semconv.ServiceVersion(cfg.ServiceVersion),
		),
	}
	if cfg.NodeHostname != "" {
		attrs = append(attrs, resource.WithAttributes(semconv.HostName(cfg.NodeHostname)))
	}
	res, err := resource.Merge(resource.Default(), resource.NewWithAttributes("", buildExtraAttrs(cfg)...))
	if err != nil {
		return nil, fmt.Errorf("merge resource: %w", err)
	}
	_ = attrs // see merge call above; resource.Option assembly is alternative path not used here
	return res, nil
}

func buildExtraAttrs(cfg Config) []attribute.KeyValue {
	out := []attribute.KeyValue{
		semconv.ServiceName(meterName),
		semconv.ServiceVersion(cfg.ServiceVersion),
	}
	if cfg.NodeHostname != "" {
		out = append(out, semconv.HostName(cfg.NodeHostname))
	}
	if cfg.InitialRole != "" {
		out = append(out, attribute.String("swarm.role", cfg.InitialRole))
	}
	return out
}
```

// NewManualHarness returns a ManualReader + a Metrics struct wired to

func NewManualHarness(tb testing.TB) (*metric.ManualReader, Metrics) {

// FindCounter collects from reader and returns the named counter's data
// points. Fails the test if the metric isn't found. Useful when you
// want to assert a specific attribute combination.
func FindCounter(tb testing.TB, reader *metric.ManualReader, name string) []metricdata.DataPoint[int64] {
	tb.Helper()

// FindHistogramCount returns the cumulative count of observations for
// the named histogram across all attribute sets. Fails the test if not
// found. Use this for "the histogram got at least one observation"
// assertions where exact attributes are tested separately.
func FindHistogramCount(tb testing.TB, reader *metric.ManualReader, name string) uint64 {
	tb.Helper()
	var rm metricdata.ResourceMetrics
	if err := reader.Collect(context.Background(), &rm); err != nil {
		tb.Fatalf("collect: %v", err)
	}
	for _, sm := range rm.ScopeMetrics {
		for _, m := range sm.Metrics {
			if m.Name != name {
				continue
			}
			switch d := m.Data.(type) {
			case metricdata.Histogram[float64]:
				var total uint64
				for _, dp := range d.DataPoints {
					total += dp.Count
				}
				return total
			}
		}
	}
	tb.Fatalf("metric %q not found or wrong type", name)
	return 0
}

// FindGauge returns the latest value of the named int64 gauge for a
// matching attribute set. Fails the test if not found.
func FindGauge(tb testing.TB, reader *metric.ManualReader, name string) []metricdata.DataPoint[int64] {
	tb.Helper()
	return findInt64Points(tb, reader, name)
}

func findInt64Points(tb testing.TB, reader *metric.ManualReader, name string) []metricdata.DataPoint[int64] {
	tb.Helper()
	var rm metricdata.ResourceMetrics
	if err := reader.Collect(context.Background(), &rm); err != nil {
		tb.Fatalf("collect: %v", err)
	}
	for _, sm := range rm.ScopeMetrics {
		for _, m := range sm.Metrics {
			if m.Name != name {
				continue
			}
			switch d := m.Data.(type) {
			case metricdata.Sum[int64]:
				return d.DataPoints
			case metricdata.Gauge[int64]:
				return d.DataPoints
			}
		}
	}
	tb.Fatalf("metric %q not found", name)
	return nil
}

// PointWithAttrs returns the data point whose attribute set matches the
// supplied key=value pairs, or fails the test.
func PointWithAttrs(tb testing.TB, points []metricdata.DataPoint[int64], kv ...string) metricdata.DataPoint[int64] {
	tb.Helper()
	if len(kv)%2 != 0 {
		tb.Fatalf("PointWithAttrs: kv must be pairs, got %d", len(kv))
	}
	for _, dp := range points {
		if matchAttrs(dp.Attributes.ToSlice(), kv) {
			return dp
		}
	}
	tb.Fatalf("no data point with attrs %v among %d points", kv, len(points))
	return metricdata.DataPoint[int64]{}
}

func matchAttrs(have []attribute.KeyValue, want []string) bool {
	for i := 0; i < len(want); i += 2 {
		key := want[i]
		val := want[i+1]
		found := false
		for _, kv := range have {
			if string(kv.Key) == key && kv.Value.AsString() == val {
				found = true
				break
			}
		}
		if !found {
			return false
		}
	}
	return true
}

package api

import (
	"net/http"
	"strconv"
	"time"

	"github.com/matchory/swarm-auto-drain/internal/metrics"
	"go.opentelemetry.io/otel/attribute"
	otelmetric "go.opentelemetry.io/otel/metric"
)

// knownRoutes is the fixed allowlist of route templates. Unknown paths
// bucket to "other" so cardinality stays bounded regardless of probe
// or attack traffic.
//
//nolint:gochecknoglobals // immutable lookup table; const-equivalent for maps
var knownRoutes = map[string]string{
	"/healthz":         "/healthz",
	"/role":            "/role",
	"/managers/health": "/managers/health",
	"/lock":            "/lock",
	"/drain":           "/drain",
	"/active":          "/active",
	"/metrics":         "/metrics",
}

// Metrics wraps next with RED-style counter and histogram recording.
// Install OUTSIDE AuditLog so the middleware overhead is itself
// measured by the histogram.
func Metrics(next http.Handler, m metrics.HTTPMetrics) http.Handler {
	return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		sr := &statusRecorder{ResponseWriter: w, status: http.StatusOK}
		start := time.Now()
		next.ServeHTTP(sr, r)
		dur := time.Since(start).Seconds()

		route := knownRoutes[r.URL.Path]
		if route == "" {
			route = "other"
		}
		baseAttrs := []attribute.KeyValue{
			attribute.String("method", r.Method),
			attribute.String("route", route),
		}
		m.Requests.Add(r.Context(), 1, otelmetric.WithAttributes(
			append(baseAttrs, attribute.String("status", strconv.Itoa(sr.status)))...,
		))
		m.Duration.Record(r.Context(), dur, otelmetric.WithAttributes(baseAttrs...))
	})
}
```

`statusRecorder` already exists in `internal/api/audit.go` — reuse it (it's package-private and we're in the same package). No need to redefine.

func (s *ConfigStore) Release(ctx context.Context, holder string) error {
	existing, err := s.loadCurrent(ctx)
	if errors.Is(err, ErrNoLock) {
		s.Metrics.Release.Add(ctx, 1, otelmetric.WithAttributes(attribute.String("outcome", "not_found")))
		return ErrNoLock
	}
	if err != nil {
		s.Metrics.Release.Add(ctx, 1, otelmetric.WithAttributes(attribute.String("outcome", "error")))
		return fmt.Errorf("inspect: %w", err)
	}
	if existing.Holder != holder {
		s.Metrics.Release.Add(ctx, 1, otelmetric.WithAttributes(attribute.String("outcome", "not_holder")))
		return ErrNotHolder
	}
	if removeErr := s.Docker.ConfigRemove(ctx, s.ConfigName); removeErr != nil && !isNotFound(removeErr) {
		s.Metrics.Release.Add(ctx, 1, otelmetric.WithAttributes(attribute.String("outcome", "error")))
		return fmt.Errorf("remove: %w", removeErr)
	}
	s.recordRelease(ctx)
	return nil
}

func (s *ConfigStore) markHeld(start time.Time) {
	s.holdMu.Lock()
	s.holdStart = start
	s.holdMu.Unlock()
}

func (s *ConfigStore) recordRelease(ctx context.Context) {
	s.holdMu.Lock()
	start := s.holdStart
	s.holdStart = time.Time{}
	s.holdMu.Unlock()
	if !start.IsZero() {
		s.Metrics.Hold.Record(ctx, time.Since(start).Seconds())
	}
	s.Metrics.Release.Add(ctx, 1, otelmetric.WithAttributes(attribute.String("outcome", "released")))
	s.Metrics.Held.Record(ctx, 0)
}
```

> **Note on `holdStart`:** the spec calls `lock_held` "1 if this replica currently holds a lock, 0 otherwise" — event-driven, not derived from store inspection. Initialising it via `markHeld` keeps drift bounded to within-process state, which is what the spec describes. Across replica restarts the gauge will be 0 even if a lock is actually held by this node, but the next acquire/release will correct it. This matches spec §10's "default to event-driven; revisit if drift observed."

	start := time.Now()
	defer func() {
		d.Metrics.Duration.Record(ctx, time.Since(start).Seconds())
		d.Metrics.TasksRemaining.Record(ctx, 0)
	}()

	if err := d.Docker.NodeUpdate(ctx, node, "drain"); err != nil {
		d.recordDrainOutcome(ctx, "error")
		return Result{}, fmt.Errorf("set drain: %w", err)
	}
	deadline := start.Add(timeout)
	for {
		count, err := d.Docker.NodeTaskCount(ctx, node)
		if err != nil {
			d.recordDrainOutcome(ctx, "error")
			return Result{}, fmt.Errorf("poll tasks: %w", err)
		}
		d.Metrics.TasksRemaining.Record(ctx, int64(count))
		if count == 0 {
			d.recordDrainOutcome(ctx, "ok")
			return Result{TasksRemaining: 0, Elapsed: time.Since(start), TimedOut: false}, nil
		}
		if time.Now().After(deadline) {
			d.recordDrainOutcome(ctx, "timeout")
			return Result{TasksRemaining: count, Elapsed: time.Since(start), TimedOut: true}, nil
		}
		select {
		case <-ctx.Done():
			d.recordDrainOutcome(ctx, "error")
			return Result{}, fmt.Errorf("drain context: %w", ctx.Err())
		case <-time.After(pollInterval):
		}
	}
}

func bindListeners(
	ctx context.Context, cfg *config.Config, overlay, host, metricsH http.Handler,
) ([]api.ListenerSpec, error) {
	var lc net.ListenConfig
	tcpLn, err := lc.Listen(ctx, "tcp", cfg.ListenAddr)

	unixLn, err := openUnixListener(ctx, cfg.UnixSocketPath, cfg.UnixSocketMode, cfg.UnixSocketGID)

	}
	specs := []api.ListenerSpec{
		{Name: "overlay-tcp", Listener: tcpLn, Handler: overlay},
		{Name: "host-unix", Listener: unixLn, Handler: host, ConnContext: unixPeerCredConnContext},
	}

## What this plan deliberately does NOT do

- No Grafana dashboards or alert rules — those belong in consumers' monitoring repos.
- No new auth on `/metrics` — access control is the operator's listener-bind choice.
- No process/CPU/RSS metrics beyond Go runtime — the host node exporter covers those.
- No OTLP gRPC exporter — HTTP is sufficient and avoids the extra dep tree.
- No trace SDK — the request flow is too shallow for tracing to pay off.
- No per-lock-id or per-source-IP labels — unbounded cardinality.
- No backward-compatible alias of old metric names — there are no old metric names.
